% Section rule: spans from the section-title edge to the right text margin.
% Derived from \textwidth and \sectionwidth so it tracks any margin change
% and is identical in every section.
\newcommand{\cvrule}{%
  \hspace*{-\sectionwidth}%
  \rule[2mm]{\dimexpr\textwidth+\sectionwidth\relax}{.25pt}}

% Section heading + rule.  \needspace keeps a heading (and its first entry)
% from being stranded at the foot of a page once breaks are automatic.
\newcommand{\cvsection}[1]{%
  \needspace{4\baselineskip}%
  \section{\textbf{#1}}%
  \cvrule}

% Numbered entry list whose items may run to several lines: \interlinepenalty
% forbids a page break inside an entry, so a publication never splits across
% two pages.  Numbering continues automatically across page breaks.
\newenvironment{cventries}%
  {\begin{enumerate}\interlinepenalty=10000\relax}%
  {\end{enumerate}}

% mylist: a one-level enumerate used for the year-keyed lists (Appointments,
% Presentations).  Every item supplies its own \item[<year>], so the declared
% label below never actually prints -- only the leftmargin has any effect.
\newlist{mylist}{enumerate}{1}
\setlist[mylist,1]{leftmargin=.5cm, label={\alph*)}}
